1. System Overview & Architectural Rationale
The llm-society-sim framework is a modular, event-driven multi-agent simulation library engineered to model dynamic information transmission, belief evolution, and misinformation epidemiology in artificial language model networks.
To execute 405 full simulation runs (~40,500 total model turns) under strict compute budget constraints, the software architecture decouples Text Generation & Debate Execution (offloaded to high-throughput free API tiers via an asynchronous router) from Internal Belief State Probing (executed remotely on Modal cloud GPUs using vLLM logit extraction).

3. Core Module Engineering Specifications
3.1 Configuration Management (config.py)
Enforces strict typing using Pydantic models to validate agent parameterizations, topology parameters, and API rate-limit quotas before execution.

```
import os
from pydantic import BaseModel, Field
from typing import List, Optional

class AgentConfig(BaseModel):
    model_id: str
    provider: str  # Options: "groq", "cerebras", "gemini", "openrouter"
    temperature: float = 0.7
    system_prompt_override: Optional[str] = None

class SimConfig(BaseModel):
    num_agents: int = 20
    num_rounds: int = 5
    context_token_budget: int = 2000
    topologies: List[str] = Field(default_factory=lambda: ["complete", "erdos_renyi", "watts_strogatz"])
    
    # Provider Concurrency Limits (Requests Per Minute)
    groq_rpm: int = 30
    gemini_rpm: int = 15
    cerebras_rpm: int = 30

    # Experimental Cohort Presets
    homogeneous_model: AgentConfig = AgentConfig(model_id="llama-3.1-8b-instant", provider="groq")
    heterogeneous_cohort: List[AgentConfig] = Field(default_factory=lambda: [
        AgentConfig(model_id="llama-3.1-8b-instant", provider="groq"),
```

3.2 Network Topology & Population Generator (graph.py)
Constructs directed communicative graphs using NetworkX and attaches model configuration attributes to node metadata.

```
import networkx as nx
from config import SimConfig

class SocietyGraph:
    def __init__(self, config: SimConfig):
        self.config = config

    def build_topology(self, topology_type: str) -> nx.DiGraph:
        N = self.config.num_agents
        if topology_type == "complete":
            G = nx.complete_graph(N, create_using=nx.DiGraph)
        elif topology_type == "erdos_renyi":
            G = nx.erdos_renyi_graph(N, p=0.2, seed=42, directed=True)
        elif topology_type == "watts_strogatz":
            undirected_G = nx.watts_strogatz_graph(N, k=4, p=0.1, seed=42)
            G = undirected_G.to_directed()
        else:
            raise ValueError(f"Unknown topology type: {topology_type}")
        
        G.graph["type"] = topology_type
        return G

    def assign_population(self, G: nx.DiGraph, population_type: str) -> nx.DiGraph:
        nodes = list(G.nodes())
        if population_type == "homogeneous":
            for node in nodes:
                G.nodes[node]["agent_config"] = self.config.homogeneous_model
        elif population_type == "heterogeneous":
            cohort = self.config.heterogeneous_cohort
            for i, node in enumerate(nodes):
                G.nodes[node]["agent_config"] = cohort[i % len(cohort)]
        else:
            raise ValueError(f"Unknown population type: {population_type}")
            
        G.graph["population_type"] = population_type
        return G
```


3.3 Bounded Context Operator M_phi (memory.py)
Prevents context window overflow and models recency bias by truncating older conversation turns while keeping system prompts pinned.

```
import tiktoken

class BoundedMemoryManager:
    """Implements context memory operator M_phi(C_i, K)."""
    
    def __init__(self, token_budget: int = 2000):
        self.budget = token_budget
        self.encoder = tiktoken.get_encoding("cl100k_base")

    def count_tokens(self, text: str) -> int:
        return len(self.encoder.encode(text))

    def prune_context(self, system_prompt: str, context_history: list[dict]) -> list[dict]:
        """Keep system prompt fixed; retain only recent incoming messages fitting token budget K."""
        system_tokens = self.count_tokens(system_prompt)
        available_budget = self.budget - system_tokens
        
        pruned_history = []
        accumulated_tokens = 0
        
        for message in reversed(context_history):
            msg_tokens = self.count_tokens(message["content"])
            if accumulated_tokens + msg_tokens <= available_budget:
                pruned_history.insert(0, message)
                accumulated_tokens += msg_tokens
            else:
                break
                
        return pruned_history
```


3.4 Async Rate-Limited Router (router.py)
Manages concurrency across free-tier APIs (Groq, Cerebras, Google AI Studio) using asyncio.Semaphore and exponential backoff to ensure zero downtime.

```
import asyncio
import os
import httpx
from config import AgentConfig

class AsyncLLMRouter:
    def __init__(self):
        self.semaphores = {
            "groq": asyncio.Semaphore(5),
            "gemini": asyncio.Semaphore(2),
            "cerebras": asyncio.Semaphore(5)
        }

    async def generate_response(self, agent_config: AgentConfig, system_prompt: str, messages: list[dict]) -> str:
        provider = agent_config.provider
        semaphore = self.semaphores.get(provider, asyncio.Semaphore(3))
        
        async with semaphore:
            for attempt in range(3):
                try:
                    if provider == "groq":
                        return await self._call_groq(agent_config.model_id, system_prompt, messages)
                    elif provider == "gemini":
                        return await self._call_gemini(agent_config.model_id, system_prompt, messages)
                    elif provider == "cerebras":
                        return await self._call_cerebras(agent_config.model_id, system_prompt, messages)
                except Exception:
                    if attempt == 2:
                        return "[Agent Error: Latency drop]"
                    await asyncio.sleep(2 ** attempt)

    async def _call_groq(self, model: str, system_prompt: str, messages: list[dict]) -> str:
        url = "https://api.groq.com/openai/v1/chat/completions"
        headers = {"Authorization": f"Bearer {os.getenv('GROQ_API_KEY')}"}
        payload = {
            "model": model,
            "messages": [{"role": "system", "content": system_prompt}] + messages,
            "temperature": 0.7
        }
        async with httpx.AsyncClient(timeout=30.0) as client:
            resp = await client.post(url, json=payload, headers=headers)
            resp.raise_for_status()
            return resp.json()["choices"][0]["message"]["content"]

    async def _call_gemini(self, model: str, system_prompt: str, messages: list[dict]) -> str:
        url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent?key={os.getenv('GEMINI_API_KEY')}"
        contents = [{"role": "user", "parts": [{"text": f"System Instruction: {system_prompt}\nContext:\n" + str(messages)}]}]
        async with httpx.AsyncClient(timeout=30.0) as client:
            resp = await client.post(url, json={"contents": contents})
            resp.raise_for_status()
            return resp.json()["candidates"][0]["content"]["parts"][0]["text"]

    async def _call_cerebras(self, model: str, system_prompt: str, messages: list[dict]) -> str:
        url = "https://api.cerebras.ai/v1/chat/completions"
        headers = {"Authorization": f"Bearer {os.getenv('CEREBRAS_API_KEY')}"}
        payload = {
            "model": model,
            "messages": [{"role": "system", "content": system_prompt}] + messages,
            "temperature": 0.7
        }
        async with httpx.AsyncClient(timeout=30.0) as client:
            resp = await client.post(url, json=payload, headers=headers)
            resp.raise_for_status()
            return resp.json()["choices"][0]["message"]["content"]
```


3.5 Serverless Modal vLLM Prober (modal_prober.py)
Runs vLLM inference on Modal cloud GPUs (L4 instances @ ~$0.80/hr) to extract token log-probabilities for exact logit probe computation.

```
import modal

app = modal.App("llm-society-logit-prober")
vllm_image = modal.Image.debian_slim().pip_install("vllm", "torch", "transformers")

@app.function(image=vllm_image, gpu="L4", timeout=600)
def extract_logit_belief_batch(agent_contexts: list[str], fact: str) -> list[float]:
    """Batch processes contexts on Modal GPU to extract logit probability P(True) / [P(True) + P(False)]."""
    from vllm import LLM, SamplingParams
    import torch

    llm = LLM(model="meta-llama/Meta-Llama-3.1-8B-Instruct")
    sampling_params = SamplingParams(temperature=0.0, max_tokens=1, logprobs=20)
    prompts = [f"{ctx}\nQuestion: Is the claim '{fact}' True or False? Answer:" for ctx in agent_contexts]
    
    outputs = llm.generate(prompts, sampling_params)
    results = []
    
    for output in outputs:
        logprobs = output.outputs[0].logprobs[0]
        p_true, p_false = 0.0, 0.0
        for token_id, logprob_obj in logprobs.items():
            text = logprob_obj.decoded_token.strip().lower()
            if text == "true":
                p_true = torch.exp(torch.tensor(logprob_obj.logprob)).item()
            elif text == "false":
                p_false = torch.exp(torch.tensor(logprob_obj.logprob)).item()
        results.append(p_true / (p_true + p_false + 1e-8))
        
    return results
```
